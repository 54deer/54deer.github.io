    容器基础

    容器是镜像的运行时

    Docker 容器存在的意义就是为了运行容器中的应用，对外提供服务，所以启动容器的目的就是启动运行该容器中的应用。

    容器中的应用运行完毕后，容器就会自动终止。如果不想让容器启动后立即终止运行，则就需要使应用处于与交互状态或等待状态

    // 删除容器
docker rm <CONTAINER> ...
docker rm $(docker ps -aq) 
-f     强制删除，包括运行中容器

// 启动容器
docker start <CONTAINER>
// 重启容器
docker restart <CONTAINER>

// 从镜像创建但不运行容器(不支持-d, --detach选项)
docker create <IMAGE>
 
// 从镜像创建并运行容器
docker run <IMAGE>
-d, --detach                     后台运行（运行后不进入应用cmd）
-e, --env list                   设置环境变量
--env-file list                  从指定文件读取环境变量
-p, --publish (h_port):(c_port)  端口映射（宿主机:容器）
-P, --publish-all                随机映射可用端口 
--name <namae>	                 给容器起名字
-i                               保持stdin打开
-t	                             分配一个伪终端
-v	                             挂载目录/卷	
--network	                     指定网络	
--restart                        重启策略	
--rm	                         退出后自动删除容器	
-w                               指定进入容器工作目录	
-u	                             指定运行用户
docker run -d --name myredis1 -p 8888:6379 redis // 不创建交互终端，后台运行
docker run -it --name myredis1 -p 8888:6379 redis // 创建并进入交互终端
docker run -dit --name myredis1 -p 8888:6379 redis // 创建但不进入交互终端，后台运行

// 以新进程的方式进入运行中容器(需指定COMMAND,如reids-cli,一般带-it开启伪终端)
// 使用exit只终止当前进程(如客户端进程)，不终止服务进程
docker exec <CONTAINER> <COMMAND>
-i                               保持stdin打开
-t	                             分配一个伪终端
-u	                             指定运行用户
-d, --detach                     后台运行（运行后不进入应用cmd）
-e, --env list                   设置环境变量
--env-file list                  从指定文件读取环境变量

docker exec -it myredis1 redis-cli // 进入redis客户端交互终端

// 进入运行中容器当前的进程(exit退出会杀死容器，默认开启stdin，但能否使用取决于容器第一次run时是否指定-it,以及run命令使用的COMMAND是否支持交互)
docker attach <CONTAINER>

// 暂停容器对外服务
docker pause <CONTAINER>
// 解除暂停容器对外服务
docker unpause <CONTAINER>

// 退出不停止容器进程
Ctrl + P + Q

// 退出并停止容器进程
exit 

// 优雅停止容器
docker stop <CONTAINER> ...
docker stop $(docker ps -aq) 

// 强制停止容器
docker kill <CONTAINER> ...
docker stop $(docker ps -aq) 


// 查看容器
docker ps
-a                     查看所有容器(up/create/exited)
-f, --filter filter    过滤结果
-n, --last int         指定显示容器数(-1是所有)
-l, --latest           显示最新的容器
-q, --quiet            只显示容器ID
-s, --size             显示占用空间大小

docker ps -aq // 查看所有容器ID

// 查看运行中容器的进程
docker top <CONTAINER>

// 查看容器的日志
docker logs <CONTAINER>
--details          显示细节
--since string     Show logs since timestam or relative
-n                 显示日志数
--until string     Show logs before a timestamp or relative
-f, --follow       动态查看日志


// 容器与宿主机之间文件传递
// 容器->宿主机
docker cp [OPTIONS] CONTAINER:SRC_PATH  DEST_PATH
// 宿主机->容器
docker cp [OPTIONS] SRC_PATH  CONTAINER:DEST_PATH


// 容器打包为新镜像(以将一个容器文件系统的当前快照生成为一个新的镜像，[REPOSITORY[:TAG]]不指定为悬虚镜像)
docker commit <CONTAINER> [REPOSITORY[:TAG]]
-a, --author string    Author
-c, --change list      Apply Dockerfile to the created image
-m, --message string   Commit message

// 导出容器(将一个容器的文件系统导出为 tar 镜像的文件)
docker export <CONTAINER>
 -o, --output string   指定导出文件，否则stdout
 
// 导出容器(将file|URL|-导入为一个镜像，[REPOSITORY[:TAG]]不指定为悬虚镜像)
 docker import  file|URL|-  [REPOSITORY[:TAG]]
-c, --change list       Apply Dockerfile to the created image
-m, --message string   Commit message

    docker commit，export 与 save三者的区分，import 与 load的区分：

    • commit 的产物是镜像（还在 Docker 里），export/save 的产物是文件（tar 包）
    • export 是扁平化的（单层），体积更小，但容器相关的元数据全丢
    • save 是分层的，完整保留，体积更大

    Commands:
  df          Show docker disk usage
  events      Get real time events from the server
  info        Display system-wide information
  prune       Remove unused data
  
// 查看磁盘占用率
docker system df
-v    详细的占用情况 

// 查看在 docker 上所发生的所有事件/命令
docker system events == docker events

// 查看dockerClient与 docker Server 等的详情信息
docker system info == docker info

// 删除 docker 中的无用数据，已经停止的容器、没有任何连接容器的网络、悬空镜像及其构建缓存
docker system prune
